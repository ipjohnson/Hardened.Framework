namespace Hardened.Templates.RazorBlade
{
    public abstract class HardenedHtmlTemplate<TModel> : RazorBlade.HtmlTemplate, Hardened.Requests.Abstract.Outputs.IHardenedResponseOutput, Hardened.Requests.Abstract.Outputs.IHardenedResponseOutput<TModel>
    {
        protected HardenedHtmlTemplate() { }
        public virtual string ContentType { get; }
        protected Hardened.Requests.Abstract.Execution.IExecutionContext Context { get; }
        public TModel Model { get; }
        public bool SupportsContentType(string? accept, Hardened.Requests.Abstract.Execution.IExecutionContext context) { }
        public System.Threading.Tasks.Task WriteOutput(Hardened.Requests.Abstract.Execution.IExecutionContext context) { }
    }
    [Hardened.Requests.Abstract.Templates.TemplateBase(typeof(Hardened.Templates.RazorBlade.HardenedHtmlTemplate<TModel>))]
    [Hardened.Requests.Abstract.Templates.TemplateContentType("text/html; charset=utf-8")]
    public sealed class RazorTemplates
    {
        public RazorTemplates() { }
    }
}
